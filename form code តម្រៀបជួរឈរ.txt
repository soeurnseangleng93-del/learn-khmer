      function handleFormSubmit(formObject) {
        event.preventDefault();
        var btn = document.getElementById("submitBtn");
        btn.disabled = true;
        btn.innerHTML = "កំពុងរក្សាទុក...";

        google.script.run
          .withSuccessHandler(function(response) {
            alert('រក្សាទុកទិន្នន័យបានជោគជ័យ!');
            document.getElementById("studentForm").reset();
            btn.disabled = false;
            btn.innerHTML = "រក្សាទុកទិន្នន័យ";
          })
          .withFailureHandler(function(error) {
            alert('មានបញ្ហា៖ ' + error.message);
            btn.disabled = false;
            btn.innerHTML = "រក្សាទុកទិន្នន័យ";
          })
          .processForm(formObject);
      }
    </script>

១. ផ្នែក <head> និង <style> (ការរចនារូបរាង CSS)
<meta name="viewport" ...>: បញ្ជាឱ្យកម្មវិធីរុករក (Browser) នៅលើទូរសព្ទស្គាល់ និងលៃតម្រូវទំហំទំព័រឱ្យត្រូវបាសជាមួយអេក្រង់ស្មាតហ្វូនដោយស្វ័យប្រវត្តិ។

font-size: 16px; និង padding: 12px 14px;: កំណត់ទំហំអក្សរ និងប្រអប់បញ្ចូលទិន្នន័យឱ្យធំល្មម (មិនតូចពេក) ដើម្បីឱ្យងាយស្រួលមើល និងចុចបំពេញដោយម្រាមដៃនៅលើទូរសព្ទ។

.form-container: ជាប្រអប់ធំពណ៌សព័ទ្ធជុំវិញ Form ទាំងអស់ ដោយកំណត់ width: 100% ដើម្បីឱ្យវាលាតសន្ធឹងពេញទំហំអេក្រង់ទូរសព្ទយ៉ាងស្រស់ស្អាត។

អន្តរកម្ម input:focus: នៅពេលអ្នកចុចចូលក្នុងប្រអប់ណាមួយ ស៊ុមរបស់ប្រអប់នឹងប្តូរទៅជាពណ៌ខຽវ (#1a73e8) ដើម្បីឱ្យដឹងថាអ្នកកំពុងវាយអក្សរនៅត្រង់ណា។

២. ផ្នែក <body> (โครงสร้าง Form)
<div class="form-container">: ផ្ទុកចំណងជើង និងទម្រង់ Form ទាំងមូល។

<form id="studentForm" onsubmit="handleFormSubmit(this)">: កំណត់ថាពេលណាដែលអ្នកចុចប៊ូតុង "រក្សាទុកទិន្នន័យ" វានឹងកុំឱ្យទំព័រ Refresh ភ្លាមៗ ប៉ុន្តែនឹងហៅមុខងារ JavaScript ឈ្មោះ handleFormSubmit មកដំណើរការជំនួសវិញ។

<fieldset> និង <legend>: ប្រើសម្រាប់បែងចែកក្រុមព័ត៌មានដាច់ដោយឡែកពីគ្នា ដូចជា "ទីកន្លែងកំណើត" និង "អាសយដ្ឋានបច្ចុប្បន្ន" ដើម្បីឱ្យ Form មើលទៅមានសោភ័ណភាព និងមានរបៀបរៀបរយ។

៣. ផ្នែក <script> (កូដ JavaScript ប្រតិបត្តិការ)
event.preventDefault();: បញ្ឈប់ការបញ្ជូនទិន្នន័យបែបធម្មតា (ដែលធ្វើឱ្យទំព័រដូរចេញឬ Refresh) ដើម្បីឱ្យយើងអាចផ្ញាទិន្នន័យទៅកាន់ Google Sheets ដោយរលូន។

ប៊ូតុងផ្ទុកបណ្ដោះអាសន្ន (btn.disabled = true;): ពេលអ្នកចុចរក្សាទុក ប៊ូតុងនឹងប្តូរអត្ថបទទៅជា "កំពុងរក្សាទុក..." និងបិទមិនឱ្យចុចซ้ำ ដើម្បីការពារការបញ្ជូនទិន្នន័យស្ទួនគ្នា។

google.script.run: ជាស្ពានតភ្ជាប់រវាងកូដ HTML (Frontend) និងកូដ Google Apps Script (Backend ក្នុង Code.gs) ដើម្បីយកទិន្នន័យដែលអ្នកបានបំពេញទៅសរសេរដាក់ចូលក្នុង Google Sheets របស់អ្នក។

.withSuccessHandler(...): ដំណើរការពេលរក្សាទុកបានសម្រេច (បង្ហាញសារជោគជ័យ និងសម្អាត Form ដើមវិញ)។

.withFailureHandler(...): ដំណើរការពេលមានបញ្ហាខុសប្រក្រតីកើតឡើង (បង្ហាញសារ Error ប្រាប់អ្នក)។
